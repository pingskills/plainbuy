cmake_minimum_required(VERSION 3.21)
project(plainbuy VERSION 0.1.0 DESCRIPTION "A small Bitcoin buy app for CoinSpot" HOMEPAGE_URL "https://github.com/pingskills/plainbuy" LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(GNUInstallDirs)
option(PLAINBUY_BUILD_TESTS "Build tests" ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Network Qml Quick QuickControls2)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SODIUM REQUIRED IMPORTED_TARGET libsodium)
qt_standard_project_setup(REQUIRES 6.5)
if(QT_KNOWN_POLICY_QTP0004)
  qt_policy(SET QTP0004 NEW)
endif()
configure_file(src/config.h.in ${CMAKE_CURRENT_BINARY_DIR}/generated/config.h @ONLY)
qt_add_library(plainbuy_core STATIC src/services/buyer.cpp src/services/buyer.h src/services/credentialstore.cpp src/services/credentialstore.h src/services/orderstore.cpp src/services/orderstore.h)
target_include_directories(plainbuy_core PUBLIC src ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_link_libraries(plainbuy_core PUBLIC Qt6::Core Qt6::Network PkgConfig::SODIUM)
qt_add_library(plainbuy_ui STATIC src/ui/theme.cpp src/ui/theme.h src/omarchy/omarchycolors.cpp src/omarchy/omarchycolors.h)
qt_add_qml_module(plainbuy_ui URI PlainBuy VERSION 1.0 OUTPUT_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/qml/PlainBuy QML_FILES qml/Main.qml)
target_link_libraries(plainbuy_ui PUBLIC plainbuy_core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2)
qt_add_executable(plainbuy src/app/main.cpp)
target_link_libraries(plainbuy PRIVATE plainbuy_core plainbuy_ui plainbuy_uiplugin)
install(TARGETS plainbuy RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES packaging/io.github.pingskills.plainbuy.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES resources/icons/io.github.pingskills.plainbuy.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/plainbuy)
if(PLAINBUY_BUILD_TESTS)
  enable_testing()
  find_package(Qt6 REQUIRED COMPONENTS Test)
  qt_add_executable(tst_buyer tests/tst_buyer.cpp)
  target_link_libraries(tst_buyer PRIVATE plainbuy_core Qt6::Test)
  add_test(NAME tst_buyer COMMAND tst_buyer)
endif()
